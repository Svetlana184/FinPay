-- =============================================================================
-- 1. ИСПРАВЛЕНИЕ СУЩЕСТВУЮЩИХ СВЯЗЕЙ
-- =============================================================================

-- 1.1. Связь bill_payments → transactions (КРИТИЧНО!)
ALTER TABLE bill_payments 
ADD CONSTRAINT fk_bill_payments_transaction 
FOREIGN KEY (transaction_id) REFERENCES transactions(id) ON DELETE SET NULL;

-- 1.2. Связь bill_payments → tax_bills (если не была создана)
ALTER TABLE bill_payments 
ADD CONSTRAINT fk_bill_payments_tax_bill 
FOREIGN KEY (tax_bill_id) REFERENCES tax_bills(id) ON DELETE SET NULL;

-- 1.3. Связь transactions → cards (для внутренних переводов)
ALTER TABLE transactions 
ADD CONSTRAINT fk_transactions_counterparty_account 
FOREIGN KEY (counterparty_account_id) REFERENCES cards(id) ON DELETE SET NULL;

-- 1.4. Связь notifications с сущностями (полиморфная через nullable FK)
ALTER TABLE notifications 
ADD COLUMN transaction_id UUID REFERENCES transactions(id) ON DELETE SET NULL,
ADD COLUMN bill_payment_id UUID REFERENCES bill_payments(id) ON DELETE SET NULL,
ADD COLUMN tax_bill_id UUID REFERENCES tax_bills(id) ON DELETE SET NULL;

-- =============================================================================
-- 2. НОВЫЕ ТАБЛИЦЫ ДЛЯ ПОЛНОЦЕННОГО БАНКОВСКОГО ПРИЛОЖЕНИЯ
-- =============================================================================

-- 2.1. Расширенные профили пользователей (KYC)
CREATE TABLE user_profiles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    email VARCHAR(255) UNIQUE,
    date_of_birth DATE,
    gender VARCHAR(10) CHECK (gender IN ('male', 'female', 'other')),
    address TEXT,
    city VARCHAR(100),
    country VARCHAR(100) DEFAULT 'Россия',
    postal_code VARCHAR(20),
    avatar_url TEXT,
    preferred_language VARCHAR(10) DEFAULT 'ru',
    preferred_currency VARCHAR(3) DEFAULT 'RUB',
    notification_settings JSONB DEFAULT '{"push": true, "email": true, "sms": false}',
    is_verified BOOLEAN DEFAULT FALSE,
    verification_level INTEGER DEFAULT 0 CHECK (verification_level BETWEEN 0 AND 3),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2.2. Избранные получатели переводов (контакты)
CREATE TABLE transfer_recipients (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    recipient_name VARCHAR(150) NOT NULL,
    recipient_account VARCHAR(50) NOT NULL, -- номер карты или счета
    recipient_bank VARCHAR(100),
    is_favorite BOOLEAN DEFAULT FALSE,
    last_transfer_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(user_id, recipient_account)
);

-- 2.3. Сберегательные счета (депозиты)
CREATE TABLE savings_accounts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    account_number VARCHAR(20) UNIQUE NOT NULL,
    account_type VARCHAR(50) NOT NULL CHECK (account_type IN ('savings', 'deposit', 'investment')),
    balance NUMERIC(15, 2) NOT NULL DEFAULT 0.00 CHECK (balance >= 0),
    currency VARCHAR(3) NOT NULL DEFAULT 'RUB',
    interest_rate NUMERIC(5, 2), -- годовая процентная ставка
    start_date DATE NOT NULL,
    end_date DATE,
    status VARCHAR(20) DEFAULT 'active' CHECK (status IN ('active', 'closed', 'frozen')),
    min_balance NUMERIC(15, 2) DEFAULT 0.00,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2.4. Транзакции по сберегательным счетам
CREATE TABLE savings_transactions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    account_id UUID NOT NULL REFERENCES savings_accounts(id) ON DELETE RESTRICT,
    transaction_type VARCHAR(50) NOT NULL CHECK (transaction_type IN ('deposit', 'withdrawal', 'interest', 'fee')),
    amount NUMERIC(15, 2) NOT NULL,
    balance_after NUMERIC(15, 2) NOT NULL,
    description TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2.5. Курсы валют
CREATE TABLE exchange_rates (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    currency_from VARCHAR(3) NOT NULL,
    currency_to VARCHAR(3) NOT NULL,
    rate NUMERIC(12, 6) NOT NULL,
    valid_from TIMESTAMP WITH TIME ZONE NOT NULL,
    valid_until TIMESTAMP WITH TIME ZONE,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(currency_from, currency_to, valid_from)
);

-- 2.6. Чат поддержки
CREATE TABLE support_conversations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    operator_id UUID REFERENCES users(id),
    subject VARCHAR(255) NOT NULL,
    status VARCHAR(20) DEFAULT 'open' CHECK (status IN ('open', 'in_progress', 'resolved', 'closed')),
    priority VARCHAR(10) DEFAULT 'normal' CHECK (priority IN ('low', 'normal', 'high', 'urgent')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    closed_at TIMESTAMP WITH TIME ZONE
);

CREATE TABLE support_messages (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    conversation_id UUID NOT NULL REFERENCES support_conversations(id) ON DELETE CASCADE,
    sender_id UUID NOT NULL REFERENCES users(id),
    message_text TEXT NOT NULL,
    attachment_url TEXT,
    is_read BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2.7. Банкоматы и отделения (геолокация)
CREATE TABLE atm_locations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    atm_code VARCHAR(50) UNIQUE NOT NULL,
    bank_name VARCHAR(100) NOT NULL,
    address TEXT NOT NULL,
    city VARCHAR(100),
    country VARCHAR(100) DEFAULT 'Россия',
    latitude DECIMAL(10, 8) NOT NULL,
    longitude DECIMAL(11, 8) NOT NULL,
    atm_type VARCHAR(20) CHECK (atm_type IN ('atm', 'branch', 'partner')),
    services JSONB DEFAULT '[]', -- ["cash_withdrawal", "balance_check", "bill_payment"]
    working_hours TEXT,
    is_24_7 BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2.8. Кредиты и займы
CREATE TABLE loans (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    loan_type VARCHAR(50) NOT NULL CHECK (loan_type IN ('personal', 'mortgage', 'auto', 'business')),
    principal_amount NUMERIC(15, 2) NOT NULL,
    remaining_balance NUMERIC(15, 2) NOT NULL,
    interest_rate NUMERIC(5, 2) NOT NULL,
    monthly_payment NUMERIC(12, 2) NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    status VARCHAR(20) DEFAULT 'active' CHECK (status IN ('active', 'paid_off', 'defaulted', 'restructured')),
    next_payment_date DATE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2.9. Платежи по кредитам
CREATE TABLE loan_payments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    loan_id UUID NOT NULL REFERENCES loans(id) ON DELETE RESTRICT,
    account_id UUID NOT NULL REFERENCES cards(id) ON DELETE RESTRICT,
    amount NUMERIC(12, 2) NOT NULL,
    payment_date DATE NOT NULL,
    status VARCHAR(20) DEFAULT 'completed' CHECK (status IN ('scheduled', 'completed', 'failed', 'late')),
    transaction_id UUID REFERENCES transactions(id) ON DELETE SET NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 4.1. Автоматическое обновление updated_at
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_users_updated_at BEFORE UPDATE ON users FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_cards_updated_at BEFORE UPDATE ON cards FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_bill_profiles_updated_at BEFORE UPDATE ON bill_profiles FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_bills_updated_at BEFORE UPDATE ON bills FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_tax_bills_updated_at BEFORE UPDATE ON tax_bills FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_user_profiles_updated_at BEFORE UPDATE ON user_profiles FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_savings_accounts_updated_at BEFORE UPDATE ON savings_accounts FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_loans_updated_at BEFORE UPDATE ON loans FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- 4.2. Автоматическое создание профиля при регистрации пользователя
CREATE OR REPLACE FUNCTION create_user_profile()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO user_profiles (user_id) VALUES (NEW.id);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_users_create_profile AFTER INSERT ON users FOR EACH ROW EXECUTE FUNCTION create_user_profile();

-- 4.3. Автоматическое создание уведомления при транзакции
CREATE OR REPLACE FUNCTION create_transaction_notification()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO notifications (user_id, type, title, body, transaction_id, channel)
    SELECT 
        c.user_id,
        'transaction',
        CASE WHEN NEW.amount > 0 THEN 'Поступление средств' ELSE 'Списание средств' END,
        CASE 
            WHEN NEW.amount > 0 THEN 'На вашу карту зачислено ' || ABS(NEW.amount) || ' ' || NEW.currency
            ELSE 'С вашей карты списано ' || ABS(NEW.amount) || ' ' || NEW.currency
        END,
        NEW.id,
        'push'
    FROM cards c WHERE c.id = NEW.account_id;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_transactions_create_notification AFTER INSERT ON transactions FOR EACH ROW EXECUTE FUNCTION create_transaction_notification();

-- 5.1. Генерация OTP-кода
CREATE OR REPLACE FUNCTION generate_otp()
RETURNS VARCHAR(6) AS $$
BEGIN
    RETURN LPAD(FLOOR(RANDOM() * 1000000)::TEXT, 6, '0');
END;
$$ LANGUAGE plpgsql;

-- 5.2. Расчет пени для просроченных налоговых счетов
CREATE OR REPLACE FUNCTION calculate_tax_penalty(p_tax_bill_id UUID)
RETURNS NUMERIC AS $$
DECLARE
    v_due_date DATE;
    v_total_amount NUMERIC;
    v_days_overdue INTEGER;
    v_penalty_rate NUMERIC := 0.0001; -- 0.01% в день (пример)
BEGIN
    SELECT due_date, total_amount INTO v_due_date, v_total_amount
    FROM tax_bills WHERE id = p_tax_bill_id;
    
    IF v_due_date IS NULL THEN
        RETURN 0;
    END IF;
    
    v_days_overdue := GREATEST(0, CURRENT_DATE - v_due_date);
    RETURN v_total_amount * v_penalty_rate * v_days_overdue;
END;
$$ LANGUAGE plpgsql;

-- 5.3. Получение актуального курса валют
CREATE OR REPLACE FUNCTION get_exchange_rate(p_from VARCHAR(3), p_to VARCHAR(3))
RETURNS NUMERIC AS $$
DECLARE
    v_rate NUMERIC;
BEGIN
    SELECT rate INTO v_rate
    FROM exchange_rates
    WHERE currency_from = p_from 
      AND currency_to = p_to 
      AND is_active = TRUE
      AND (valid_until IS NULL OR valid_until > CURRENT_TIMESTAMP)
    ORDER BY valid_from DESC
    LIMIT 1;
    
    RETURN COALESCE(v_rate, 1.0);
END;
$$ LANGUAGE plpgsql;

-- 6.1. Сводка по пользователю (включая все счета)
CREATE OR REPLACE VIEW user_full_summary AS
SELECT 
    u.id AS user_id,
    u.full_name,
    u.phone,
    up.email,
    up.is_verified,
    up.verification_level,
    (SELECT COUNT(*) FROM cards WHERE user_id = u.id AND status = 'active') AS active_cards_count,
    (SELECT COALESCE(SUM(balance), 0) FROM cards WHERE user_id = u.id AND status = 'active') AS cards_balance,
    (SELECT COUNT(*) FROM savings_accounts WHERE user_id = u.id AND status = 'active') AS savings_count,
    (SELECT COALESCE(SUM(balance), 0) FROM savings_accounts WHERE user_id = u.id AND status = 'active') AS savings_balance,
    (SELECT COUNT(*) FROM loans WHERE user_id = u.id AND status = 'active') AS active_loans_count,
    (SELECT COALESCE(SUM(remaining_balance), 0) FROM loans WHERE user_id = u.id AND status = 'active') AS total_debt,
    (SELECT COUNT(*) FROM notifications WHERE user_id = u.id AND is_read = FALSE) AS unread_notifications
FROM users u
LEFT JOIN user_profiles up ON u.id = up.user_id;

-- 6.2. История платежей с деталями
CREATE OR REPLACE VIEW bill_payments_full AS
SELECT 
    bp.id,
    bp.user_id,
    bp.amount,
    bp.currency,
    bp.status,
    bp.created_at,
    bp.completed_at,
    CASE 
        WHEN bp.bill_id IS NOT NULL THEN 'bill'
        WHEN bp.tax_bill_id IS NOT NULL THEN 'tax_bill'
    END AS payment_type,
    COALESCE(b.bill_number, tb.document_number) AS document_number,
    COALESCE(b.recipient_name, tb.tax_authority) AS recipient,
    c.card_type || ' ' || RIGHT(c.card_number, 4) AS card_masked,
    t.id AS transaction_id
FROM bill_payments bp
LEFT JOIN bills b ON bp.bill_id = b.id
LEFT JOIN tax_bills tb ON bp.tax_bill_id = tb.id
LEFT JOIN cards c ON bp.account_id = c.id
LEFT JOIN transactions t ON bp.transaction_id = t.id;

-- 6.3. Ближайшие платежи (счета и налоги)
CREATE OR REPLACE VIEW upcoming_payments AS
SELECT 
    'bill' AS type,
    id,
    bill_number AS document_number,
    recipient_name AS recipient,
    total_amount AS amount,
    currency,
    due_date,
    status
FROM bills
WHERE status = 'pending' AND due_date > CURRENT_DATE

UNION ALL

SELECT 
    'tax_bill' AS type,
    id,
    document_number,
    tax_authority AS recipient,
    total_amount,
    currency,
    due_date,
    status
FROM tax_bills
WHERE status IN ('pending', 'overdue') AND due_date > CURRENT_DATE

ORDER BY due_date ASC;
